def run(args, name, cwd=REPO, env=None, fail=True):
    merged=ENV.copy(); merged.update(env or {})
    logfile=R/(name+'.log'); logfile.parent.mkdir(parents=True,exist_ok=True)
    entry={'name':name,'argv':list(map(str,args)),'cwd':str(cwd),'env_overrides':env or {},'start':time.strftime('%Y-%m-%dT%H:%M:%S%z'),'log':str(logfile)}
    print('START '+name,flush=True)
    start=time.monotonic()
    with logfile.open('w') as f:
        p=subprocess.run(list(map(str,args)),cwd=cwd,env=merged,stdout=f,stderr=subprocess.STDOUT)
    entry.update(exit_status=p.returncode,elapsed_s=time.monotonic()-start,end=time.strftime('%Y-%m-%dT%H:%M:%S%z'))
    with (R/'commands.jsonl').open('a') as f:f.write(json.dumps(entry)+'\n')
    print('END '+name+' status='+str(p.returncode)+' elapsed='+str(round(entry['elapsed_s'],2)),flush=True)
    if p.returncode:
        print(logfile.read_text()[-12000:],flush=True)
        if fail:sys.exit(p.returncode)
    return p.returncode
